```js
// 🎮 GTA Egypt Ultimate Online Full Prototype
// 📱 React Native + Online + Offline + Egypt Map

import React, { useState, useEffect } from "react";
import {
  View,
  Text,
  Pressable,
  StyleSheet,
  ScrollView,
} from "react-native";

// 🌍 الدول
const countries = [
  { name: "Egypt 🇪🇬", color: "#27ae60", police: 3 },
  { name: "Italy 🇮🇹", color: "#3498db", police: 2 },
  { name: "UAE 🇦🇪", color: "#f1c40f", police: 4 },
  { name: "France 🇫🇷", color: "#8e44ad", police: 2 },
];

// 🏙️ محافظات مصر
const egyptCities = [
  "Cairo",
  "Giza",
  "Alexandria",
  "Tanta",
  "Mansoura",
  "Luxor",
  "Aswan",
  "Sharm El Sheikh",
  "Hurghada",
];

// 🚗 العربيات
const cars = [
  { name: "BMW", speed: 10, color: "#3498db" },
  { name: "Mercedes", speed: 8, color: "#2c3e50" },
  { name: "Audi", speed: 9, color: "#7f8c8d" },
  { name: "Ferrari", speed: 15, color: "#e74c3c" },
  { name: "Lamborghini", speed: 17, color: "#f1c40f" },
  { name: "Bugatti", speed: 20, color: "#9b59b6" },
  { name: "Tesla", speed: 13, color: "#ecf0f1" },
  { name: "Toyota", speed: 7, color: "#95a5a6" },
  { name: "Lada 2107", speed: 5, color: "#8e6e53" },
];

// 🏪 المحلات
const shops = [
  "🚗 Car Shop",
  "⛽ Gas Station",
  "🔧 Garage",
  "💰 Bank",
  "☕ Cafe",
  "🏪 Mall",
];

// 👮 أقسام الشرطة
const policeStations = [
  "👮 Cairo Police",
  "👮 Alexandria Police",
  "👮 Dubai Police",
  "👮 Rome Police",
];

// ✈️ المطارات
const airports = [
  "✈️ Cairo Airport",
  "✈️ Dubai Airport",
  "✈️ Rome Airport",
  "✈️ Paris Airport",
];

export default function App() {
  // 🚗 اللاعب
  const [player, setPlayer] = useState({
    x: 150,
    y: 300,
  });

  // 👮 الشرطة
  const [police, setPolice] = useState({
    x: 20,
    y: 20,
  });

  // 💰 فلوس
  const [money, setMoney] = useState(0);

  // 🚗 العربية الحالية
  const [carIndex, setCarIndex] = useState(0);

  // 🌍 الدولة الحالية
  const [countryIndex, setCountryIndex] = useState(0);

  // 📍 المحافظة الحالية
  const [cityIndex, setCityIndex] = useState(0);

  // 🌐 مود اللعب
  const [mode, setMode] = useState("offline");

  // 🎯 المهمة الحالية
  const [mission, setMission] = useState(
    "🚗 اجمع فلوس واهرب من الشرطة"
  );

  const car = cars[carIndex];
  const country = countries[countryIndex];

  // 🚗 حركة اللاعب
  const move = (dx, dy) => {
    setPlayer((p) => ({
      x: Math.max(0, Math.min(320, p.x + dx * car.speed)),
      y: Math.max(0, Math.min(620, p.y + dy * car.speed)),
    }));
  };

  // 👮 الشرطة الذكية
  useEffect(() => {
    const interval = setInterval(() => {
      setPolice((p) => ({
        x:
          p.x +
          (player.x > p.x
            ? country.police
            : -country.police),

        y:
          p.y +
          (player.y > p.y
            ? country.police
            : -country.police),
      }));
    }, 200);

    return () => clearInterval(interval);
  }, [player, country]);

  // 💀 الإمساك باللاعب
  useEffect(() => {
    const dist =
      Math.abs(player.x - police.x) +
      Math.abs(player.y - police.y);

    if (dist < 30) {
      setMoney(0);

      setMission("💀 الشرطة قبضت عليك!");

      setPolice({
        x: 20,
        y: 20,
      });
    }
  }, [player, police]);

  // 💰 جمع الفلوس
  const collectMoney = () => {
    setMoney((m) => m + 10);

    if (money > 100) {
      setMission("🚓 الشرطة زادت المطاردة!");
    }
  };

  // ✈️ السفر
  const travel = (index) => {
    setCountryIndex(index);

    setMission(
      `✈️ سافرت إلى ${countries[index].name}`
    );
  };

  return (
    <View
      style={[
        styles.container,
        {
          backgroundColor: country.color,
        },
      ]}
    >
      {/* 🎮 العنوان */}
      <Text style={styles.title}>
        🔥 GTA Egypt Ultimate
      </Text>

      {/* 🌐 مود */}
      <Text style={styles.mode}>
        🌍 {mode.toUpperCase()}
      </Text>

      {/* 💰 فلوس */}
      <Text style={styles.money}>
        💰 {money}
      </Text>

      {/* 🎯 المهمة */}
      <Text style={styles.mission}>
        {mission}
      </Text>

      {/* 📍 المحافظة */}
      <Text style={styles.city}>
        📍 {egyptCities[cityIndex]}
      </Text>

      {/* 🚗 اللاعب */}
      <View
        style={[
          styles.player,
          {
            left: player.x,
            top: player.y,
            backgroundColor: car.color,
          },
        ]}
      />

      {/* 👮 الشرطة */}
      <View
        style={[
          styles.police,
          {
            left: police.x,
            top: police.y,
          },
        ]}
      />

      {/* 💰 فلوس */}
      <Pressable
        style={styles.coin}
        onPress={collectMoney}
      >
        <Text style={{ fontSize: 25 }}>
          💰
        </Text>
      </Pressable>

      {/* 🎮 التحكم */}
      <View style={styles.controls}>
        <Pressable onPress={() => move(0, -1)}>
          <Text style={styles.btn}>⬆️</Text>
        </Pressable>

        <View style={{ flexDirection: "row" }}>
          <Pressable onPress={() => move(-1, 0)}>
            <Text style={styles.btn}>⬅️</Text>
          </Pressable>

          <Pressable onPress={() => move(1, 0)}>
            <Text style={styles.btn}>➡️</Text>
          </Pressable>
        </View>

        <Pressable onPress={() => move(0, 1)}>
          <Text style={styles.btn}>⬇️</Text>
        </Pressable>
      </View>

      {/* 🚗 الجراج */}
      <ScrollView style={styles.garage}>
        <Text style={styles.boxTitle}>
          🚗 Garage
        </Text>

        {cars.map((c, i) => (
          <Pressable
            key={i}
            onPress={() => setCarIndex(i)}
          >
            <Text style={styles.listItem}>
              {carIndex === i ? "👉 " : ""}
              {c.name}
            </Text>
          </Pressable>
        ))}
      </ScrollView>

      {/* 🌍 الدول */}
      <ScrollView style={styles.travel}>
        <Text style={styles.boxTitle}>
          ✈️ Travel
        </Text>

        {countries.map((c, i) => (
          <Pressable
            key={i}
            onPress={() => travel(i)}
          >
            <Text style={styles.listItem}>
              🌍 {c.name}
            </Text>
          </Pressable>
        ))}
      </ScrollView>

      {/* 🏪 المحلات */}
      <ScrollView style={styles.shops}>
        <Text style={styles.boxTitle}>
          🏪 Shops
        </Text>

        {shops.map((s, i) => (
          <Text
            key={i}
            style={styles.listItem}
          >
            {s}
          </Text>
        ))}
      </ScrollView>

      {/* 👮 الشرطة */}
      <ScrollView style={styles.policeBox}>
        <Text style={styles.boxTitle}>
          👮 Police
        </Text>

        {policeStations.map((p, i) => (
          <Text
            key={i}
            style={styles.listItem}
          >
            {p}
          </Text>
        ))}
      </ScrollView>

      {/* ✈️ المطارات */}
      <ScrollView style={styles.airports}>
        <Text style={styles.boxTitle}>
          ✈️ Airports
        </Text>

        {airports.map((a, i) => (
          <Text
            key={i}
            style={styles.listItem}
          >
            {a}
          </Text>
        ))}
      </ScrollView>

      {/* 🌐 Online / Offline */}
      <View style={styles.modeSwitch}>
        <Pressable
          onPress={() => setMode("offline")}
        >
          <Text style={styles.switchBtn}>
            📴 Offline
          </Text>
        </Pressable>

        <Pressable
          onPress={() => setMode("online")}
        >
          <Text style={styles.switchBtn}>
            🌍 Online
          </Text>
        </Pressable>
      </View>
    </View>
  );
}
```
